import dht
from machine import Pin
import time

sensor = dht.DHT22(Pin(4))

print("Starting DHT22 sensor readings...")

while True:
    try:
        sensor.measure()
        
        temp = sensor.temperature()
        hum = sensor.humidity()
        
        print(f"Temperature: {temp:.1f}°C  |  Humidity: {hum:.1f}%")
        
    except OSError as e:
        print("Failed to read sensor. The data line is still floating or silent!")
    time.sleep(2)
